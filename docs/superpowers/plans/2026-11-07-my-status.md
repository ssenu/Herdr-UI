# 시스템 모니터 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 하단 바 오른쪽에 이 PC의 CPU·RAM·GPU 사용량을 보여 주고, 마우스를 올리면 최근 5분 그래프 대시보드를 띄운다.

**Architecture:** `my-status`는 기본 `workspace-status` 복제본으로, 5초마다 PowerShell 한 번으로 값을 읽고 메모리에 기록을 쌓는다. `my-status-dash`는 대시보드 오버레이이며 값을 읽지 않고 `my-status`가 계산해 준 행(`rows`)만 그린다. 판단 로직은 전부 `mods/my-status/logic.js`에 두고 `node --test`로 검증한다.

**Tech Stack:** Herdr mods (JS, `gpui-kit`, `gpui-base`, `herdr`), PowerShell CIM, `node:test`.

## Global Constraints

- 범위: 이 PC의 CPU·RAM·GPU와 최근 5분 그래프. Windows만. 원격 머신 수치, 디스크·네트워크, 기록 저장은 하지 않는다.
- 읽기 간격 5초, 기록 60칸, 호버 대기 0.3초, 강조 기준 90% 이상.
- 하단 바 형식: `CPU 13%  RAM 19.7/31.4GB  GPU 8%`, 원격 스페이스에서는 앞에 `이 PC`.
- 연속 실패 2번까지는 이전 값 유지, 3번째부터 수치를 숨기고 기록에 빈 칸.
- `cx.notify()`를 render 안에서 부르지 않는다. 사용자에게 보이는 글은 한국어.
- 도구 주의: 코드는 파일 편집 도구로 쓴다(Bash heredoc은 백슬래시를 망가뜨린다). `install.ps1`은 UTF-8 BOM을 유지한다.
- 테스트는 Git Bash에서 파일을 지정해 실행한다(`node --test tests/<file>`, 디렉터리 형식은 Node 24에서 실패).

// Pure helpers for my-status. No herdr or gpui imports, so `node --test` can load this file.

const KB_PER_GB = 1048576, HOT = 90, HIDE_AFTER = 3;

/** One PowerShell run printing "METRICS<TAB>cpu<TAB>ramUsedKb<TAB>ramTotalKb<TAB>gpu"; an unreadable value stays blank. */
export function metricsCommand() {
  const script = [
    "$ErrorActionPreference = 'Stop'",
    "$cpu = ''; $used = ''; $total = ''; $gpu = ''",
    "try { $cpu = (Get-CimInstance Win32_PerfFormattedData_PerfOS_Processor -Filter \"Name='_Total'\").PercentProcessorTime } catch {}",
    "try { $o = Get-CimInstance Win32_OperatingSystem; $total = $o.TotalVisibleMemorySize; $used = $total - $o.FreePhysicalMemory } catch {}",
    // Each engine type's total, and the busiest type: what Task Manager's GPU figure is close to.
    "try { $g = @(Get-CimInstance Win32_PerfFormattedData_GPUPerformanceCounters_GPUEngine); if ($g.Count) { " +
      "$gpu = ($g | Group-Object { $_.Name -replace '^.*engtype_', '' } | ForEach-Object { ($_.Group | Measure-Object UtilizationPercentage -Sum).Sum } | Measure-Object -Maximum).Maximum } } catch {}",
    "\"METRICS`t$cpu`t$used`t$total`t$gpu\"",
  ].join("; ");
  return { command: "powershell", args: ["-NoProfile", "-Command", script] };
}

function number(s, max) {
  if (s === undefined || String(s).trim() === "") return null;
  const n = Number(s);
  if (!Number.isFinite(n) || n < 0) return null;
  return max === undefined ? n : Math.min(n, max);
}

/** metricsCommand's output → values, or null when neither CPU nor RAM could be read. */
export function parseMetrics(stdout) {
  const line = String(stdout ?? "").split(/\r?\n/).find(l => l.startsWith("METRICS\t"));
  if (!line) return null;
  const f = line.split("\t");
  const cpu = number(f[1], 100), used = number(f[2]), total = number(f[3]), gpu = number(f[4], 100);
  const ram = used !== null && total !== null && total > 0 && used <= total;
  if (cpu === null && !ram) return null;
  return { cpu, ramUsed: ram ? used / KB_PER_GB : null, ramTotal: ram ? total / KB_PER_GB : null, gpu };
}

export const isHot = percent => typeof percent === "number" && percent >= HOT;

export function ramPercent(m) {
  return typeof m?.ramUsed === "number" && m.ramTotal > 0 ? (m.ramUsed / m.ramTotal) * 100 : null;
}

export const emptyHistory = () => ({ cpu: [], ram: [], gpu: [] });

/** Adds one sample (null: a blank) of percentages, dropping the oldest past the limit. */
export function pushHistory(history, metrics, limit = 60) {
  const add = (list, value) => [...list, value ?? null].slice(-limit);
  return { cpu: add(history.cpu, metrics?.cpu), ram: add(history.ram, ramPercent(metrics)), gpu: add(history.gpu, metrics?.gpu) };
}

/** Counts failures in a row; the figures hide from the third on. */
export function failStep(state, ok) {
  if (ok) return { fails: 0, hidden: false };
  const fails = state.fails + 1;
  return { fails, hidden: fails >= HIDE_AFTER };
}

/** The status bar's items, in order. */
export function summaryParts(metrics, remote) {
  if (!metrics) return [];
  const parts = remote ? [{ label: "이 PC", value: "", hot: false }] : [];
  if (metrics.cpu !== null) parts.push({ label: "CPU", value: `${Math.round(metrics.cpu)}%`, hot: isHot(metrics.cpu) });
  if (metrics.ramUsed !== null) parts.push({ label: "RAM", value: `${metrics.ramUsed.toFixed(1)}/${metrics.ramTotal.toFixed(1)}GB`, hot: isHot(ramPercent(metrics)) });
  if (metrics.gpu !== null) parts.push({ label: "GPU", value: `${Math.round(metrics.gpu)}%`, hot: isHot(metrics.gpu) });
  return parts;
}

/** Percentages → bar heights in px. Any value shows at least 1 px; a blank shows nothing. */
export function barHeights(values, height) {
  return values.map(v => (v === null || v === undefined ? 0 : Math.max(1, Math.round((v / 100) * height))));
}

const bars = (values, height) => {
  const heights = barHeights(values, height);
  return values.map((v, i) => ({ h: heights[i], hot: isHot(v) }));
};

/** The dashboard's rows: everything my-status-dash draws. GPU only once it has been read. */
export function dashRows(metrics, history, barHeight) {
  const percent = v => (typeof v === "number" ? `${Math.round(v)}%` : "—");
  const ram = typeof metrics?.ramUsed === "number" ? `${metrics.ramUsed.toFixed(1)} / ${metrics.ramTotal.toFixed(1)} GB` : "—";
  const rows = [
    { key: "cpu", label: "CPU", value: percent(metrics?.cpu), hot: isHot(metrics?.cpu), bars: bars(history.cpu, barHeight) },
    { key: "ram", label: "RAM", value: ram, hot: isHot(ramPercent(metrics)), bars: bars(history.ram, barHeight) },
  ];
  if (typeof metrics?.gpu === "number" || history.gpu.some(v => v !== null)) {
    rows.push({ key: "gpu", label: "GPU", value: percent(metrics?.gpu), hot: isHot(metrics?.gpu), bars: bars(history.gpu, barHeight) });
  }
  return rows;
}

/** The dashboard's height for its rows: 80 for title + footer, 60 per row (with slack so nothing clips). */
export const dashHeight = rowCount => 80 + rowCount * 60;

const MARGIN = 14, GAP = 6, EDGE = 8;

/** Where the dashboard opens: above the bar, at its right end, on screen. Null until the bar was measured. */
export function dashAnchor(bounds, size) {
  if (!bounds) return null;
  return {
    x: Math.max(EDGE, bounds.x + bounds.width - size.width - MARGIN),
    y: Math.max(EDGE, bounds.y - size.height - GAP),
  };
}

```js
// Herdr's status bar plus this PC's CPU, RAM and GPU beside the commands hint, read every 5 s. Windows only.
import { View, div, svg } from "gpui-kit";
import { h_flex } from "gpui-base";
import { spaces, agents, theme, run, window, keys, exec } from "herdr";
import { metricsCommand, parseMetrics, emptyHistory, pushHistory, failStep, summaryParts } from "./logic.js";

const READ_MS = 5000;

export default class MyStatus extends View {
  init(_props, cx) {
    this.metrics = null; this.history = emptyHistory(); this.fails = { fails: 0, hidden: false }; this.reading = false;
    if (window()?.platform !== "windows") return;
    this.read(cx);
    cx.timer.every(READ_MS, cx => this.read(cx));
  }

  read(cx) {
    if (this.reading) return;
    this.reading = true;
    cx.spawn(async cx => {
      let read = null;
      try {
        const c = metricsCommand(), r = await exec(c.command, c.args);
        if (r.code === 0) read = parseMetrics(r.stdout);
      } catch {}
      if (read) this.metrics = read;
      this.fails = failStep(this.fails, Boolean(read));
      this.history = pushHistory(this.history, this.fails.hidden ? null : this.metrics);
      this.reading = false;
      cx.notify();
    });
  }

  // The palette key as actually bound, not a hardcoded one; without a
  // binding the hint stays useful ("commands") instead of lying.
  hint(t) {
    const mac = window()?.platform === "macos";
    const mod = { cmd: mac ? "⌘" : "Ctrl+", ctrl: mac ? "⌃" : "Ctrl+", shift: mac ? "⇧" : "Shift+", alt: mac ? "⌥" : "Alt+", option: mac ? "⌥" : "Alt+" };
    const b = keys().find((k) => k.section === "global" && k.action === "app.palette");
    const parts = b && b.keys.endsWith("--") ? [...b.keys.slice(0, -2).split("-"), "-"] : b?.keys.split("-");
    const key = parts && parts.map((p) => mod[p.toLowerCase()] ?? p.toUpperCase()).join("");
    return key ? `${key}  commands` : "commands";
  }

  render() {
    const t = theme();
    const space = spaces().find(s => s.active);
    const list = agents();
    const needs = list.filter(a => a.status === "blocked").length;
    const label = (icon, text) => h_flex().gap(6).items_center().child(svg(`mods/workspace-chrome/${icon}.svg`).size(13)).child(text);
    const parts = this.fails.hidden ? [] : summaryParts(this.metrics, Boolean(space?.machine));
    return h_flex().id("my-status").size_full().px(14).gap(16).items_center().bg(t.chrome).border_t(1).border_color(t.border).text_size(11.8).text_color(t.muted)
      .child(label(space?.machine ? "server" : "monitor", space?.machine ?? "This computer"))
      .when(Boolean(space?.branch), el => el.child(label("branch", space.branch)))
      .child(h_flex().gap(6).items_center().child(div().size(6).rounded(3).bg(t.dim)).child(`${list.length} agents`)
        .when(needs > 0, el => el.child(div().text_color(t.blocked).child(`· ${needs} needs you`))))
      .child(div().flex_1())
      .when(parts.length > 0, el => el.child(h_flex().id("status-metrics").flex_none().gap(10).items_center()
        .children(parts.map(p => div().flex_none().when(p.hot, d => d.text_color(t.blocked)).child(p.value ? `${p.label} ${p.value}` : p.label)))))
      .child(div().id("palette").text_color(t.dim).on_click(() => run("app.palette")).child(this.hint(t)));
  }
}
```

```js
// The status bar's dashboard: draws the rows my-status computes (kept in shared while open). Reads nothing itself.
import { View, div } from "gpui-kit";
import { v_flex, h_flex } from "gpui-base";
import { props, shared, theme } from "herdr";

const BAR_AREA = 28;

export default class MyStatusDash extends View {
  render() {
    const t = theme(), rows = shared("my-status.dash") ?? props()?.rows ?? [];
    return v_flex().size_full().p(12).gap(8).rounded(10).overflow_hidden().bg(t.popover).border_1().border_color(t.border)
      .text_color(t.foreground).text_size(12)
      .child(div().font_semibold().child("시스템 · 이 PC"))
      .children(rows.map(r => v_flex().flex_none().gap(4)
        .child(h_flex().justify_between().items_center()
          .child(div().text_color(t.muted).child(r.label))
          .child(div().when(r.hot, d => d.text_color(t.blocked)).child(r.value)))
        .child(h_flex().h(BAR_AREA).gap(1).items_end().justify_end()
          .children(r.bars.map(b => div().flex_none().w(3).h(b.h).bg(b.hot ? t.blocked : t.accent))))))
      .child(div().text_size(11).text_color(t.dim).child("최근 5분"));
  }
}
```

```js
  rows() {
    return dashRows(this.fails.hidden ? null : this.metrics, this.history, BAR_AREA);
  }

  openDash() {
    if (!this.hovered || !this.ripe || this.dash || !this.origin || !this.size) return;
    const rows = this.rows(), size = { width: DASH_WIDTH, height: dashHeight(rows.length) };
    const at = dashAnchor({ ...this.origin, ...this.size }, size);
    if (!at) return;
    setShared(DASH_KEY, rows);
    this.dash = openOverlay({ mod: "my-status-dash", kind: "window", frame: "none", ...size, at, props: { rows } });
  }

  closeDash() {
    const id = this.dash;
    this.dash = null;
    if (!id) return;
    setShared(DASH_KEY, null);
    try { closeOverlay(id); } catch {} // a layout switch may already have closed it
  }

  hover(on, cx) {
    this.hovered = on; this.ripe = false;
    const seq = ++this.hoverSeq;
    if (!on) { this.closeDash(); return; }
    cx.timer.after(HOVER_MS, () => {
      if (seq !== this.hoverSeq) return;
      this.ripe = true;
      this.openDash();
    });
  }

  // A reload must not leave the dashboard behind.
  async close() { this.closeDash(); }

```
