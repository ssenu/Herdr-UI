# 파일 뷰어 탭 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 터미널 탭처럼 열리는 파일 뷰어·편집기 탭을 만들고, `+` 메뉴와 탭 바 드롭으로 연다.

**Architecture:** `my-viewer`는 `tab.new { mod, props }`로 만드는 탭 전체 mod로, 탭 하나에 파일 하나를 연다. 여는 쪽(`my-files-drag`, `my-new-menu`)과 뷰어가 공유하는 판단은 `viewer-open.js`에, 뷰어 판단은 `logic.js`에 둔다. mod는 자기 폴더 밖 파일을 불러올 수 없으므로 공유 파일은 복사본을 두고 테스트로 내용이 같은지 확인한다.

**Tech Stack:** Herdr mods (JS, `gpui-kit`, `gpui-base`, `herdr`, `fs/promises`), `node:test`.

## Global Constraints

- 범위: 탭 하나에 파일 하나를 여는 텍스트 뷰어·편집기. 이미지·PDF 표시, 탭 안의 파일 트리, 탭 안의 파일 탭은 하지 않는다.
- mod는 자기 폴더 밖을 import할 수 없다(`resolves outside the application directory`, 확인함). 그래서 복사본을 둔다.
- `run()`은 mod 안에서 `null`을 돌려주고, 탭 mod의 `props()`에 자기 칸·탭 ID가 없다(확인함). 그래서 여는 쪽이 고유한 임시 탭 이름을 주고, 뷰어가 그 이름으로 자기 탭을 찾은 뒤 이름을 바꾼다.
- `herdr-app check`는 `init`까지 실제로 돌린다(확인함). `fs/promises`의 `mkdir`, `writeFile`, `unlink`는 앱 폴더에서 동작한다(확인함).
- 초안 위치: `<앱 폴더>/data/my-viewer/drafts/<경로 해시>.json`, 내용 `{ path, text, base }`. 쓰기는 마지막 입력 1초 뒤. 로컬 파일만.
- 400,000자 초과는 읽기 전용(`my-files`의 `PREVIEW_LIMIT`).
- `cx.notify()`를 render 안에서 부르지 않는다. 사용자에게 보이는 글은 한국어.
- 도구 주의: 코드는 파일 편집 도구로 쓴다(Bash heredoc은 백슬래시를 망가뜨린다). `install.ps1`은 UTF-8 BOM을 유지한다.
- 테스트는 Git Bash에서 실행한다: `node --test tests/*.test.mjs`.

```js
// Shared by my-viewer, my-files-drag and my-new-menu. Mods cannot import outside their own folder,
// so each keeps a copy; tests/my-viewer-logic.test.mjs checks that the copies are identical.

export const BOUNDS_KEY = "my-tabs.bounds", OPEN_KEY = "my-viewer.open", OWN_KEY = "my-viewer.own";

const TOP_STRIP = { x: 0, y: 0, width: Infinity, height: 40 };
const inside = (b, p) => Boolean(b) && p.x >= b.x && p.x <= b.x + b.width && p.y >= b.y && p.y <= b.y + b.height;

/** What releasing a dragged row does. Until the tab bar has been measured, it is taken as the window's top strip. */
export function dropAction({ position, panel, tabs, dir }) {
  if (!position || inside(panel, position)) return "cancel";
  if (inside(tabs ?? TOP_STRIP, position)) return dir ? "folder" : "open";
  return "paste";
}

/** For comparing paths: "/" separators; Windows drive paths ignore case. */
export function normKey(path) {
  const s = String(path ?? "").replaceAll("\\", "/");
  return /^[A-Za-z]:/.test(s) ? s.toLowerCase() : s;
}

export const openKey = (on, path) => `${on ?? ""}|${normKey(path)}`;

/** The tab registered for key, if it still exists. */
export function liveTab(open, key, spaces) {
  const id = open?.[key];
  if (!id) return null;
  for (const s of spaces) if (s.tabs.some(t => t.id === id)) return { space: s.id, tab: id };
  return null;
}

export const newToken = (now, random) => `${now.toString(36)}${Math.floor(random * 1e9).toString(36)}`;

/** tab.new's arguments. The viewer finds its own tab by the unique tabName, then renames it. */
export function tabArgs({ path, on, root, space }, token) {
  const tabName = `파일 뷰어 ${token}`;
  return { name: tabName, space, mod: "my-viewer", props: { path, on, root, token, tabName } };
}

/** The commands opening this file: the tab already showing it, or a new one. */
export function openCommands(where, open, spaces, token) {
  const live = where.path ? liveTab(open, openKey(where.on, where.path), spaces) : null;
  if (!live) return [["tab.new", tabArgs(where, token)]];
  const shown = spaces.find(s => s.active)?.id === live.space;
  return [...(shown ? [] : [["space.select", { id: live.space }]]), ["tab.select", { id: live.tab }]];
}
```

```js
// Pure helpers for my-viewer. No herdr or gpui imports, so `node --test` can load this file.
import { normKey } from "./viewer-open.js";
import { isMarkdown, sameText } from "./files-logic.js";

const BINARY = /\.(png|jpe?g|gif|bmp|webp|ico|tiff?|heic|avif|pdf|zip|7z|rar|gz|tgz|bz2|xz|tar|exe|dll|so|dylib|bin|msi|dmg|iso|class|jar|woff2?|ttf|otf|eot|mp3|wav|flac|ogg|mp4|mov|avi|mkv|webm|docx?|xlsx?|pptx?|hwpx?|sqlite|db)$/i;

/** How to show a file. text: null means it hasn't been read, so only the name decides. */
export function viewKind(path, text) {
  if (BINARY.test(path ?? "") || (typeof text === "string" && text.includes("\0"))) return "binary";
  return isMarkdown(path) ? "markdown" : "text";
}

/** A short stable hash (cyrb53 variant) as 16 hex digits. */
export function textHash(text) {
  let h1 = 0xdeadbeef, h2 = 0x41c6ce57;
  for (let i = 0; i < text.length; i++) {
    const c = text.charCodeAt(i);
    h1 = Math.imul(h1 ^ c, 2654435761);
    h2 = Math.imul(h2 ^ c, 1597334677);
  }
  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
  return (h2 >>> 0).toString(16).padStart(8, "0") + (h1 >>> 0).toString(16).padStart(8, "0");
}

/** The draft file's name for a path, without its extension. */
export const draftName = path => textHash(normKey(path));

/** A draft file's text → { path, text, base }, or null unless complete. */
export function parseDraft(text) {
  if (typeof text !== "string") return null;
  try {
    const d = JSON.parse(text);
    return typeof d?.path === "string" && typeof d.text === "string" && typeof d.base === "string" ? { path: d.path, text: d.text, base: d.base } : null;
  } catch { return null; }
}

/**
 * What the editor starts with. disk: the file's text, null when it's gone.
 * dirty: differs from the disk. conflict: the disk changed since the draft began. discard: the draft is useless.
 */
export function restoreState(draft, path, disk) {
  const clean = { text: disk, dirty: false, conflict: false, discard: false };
  if (!draft || normKey(draft.path) !== normKey(path)) return clean;
  if (disk === null) return { text: draft.text, dirty: true, conflict: true, discard: false };
  if (sameText(draft.text, disk)) return { ...clean, discard: true };
  return { text: draft.text, dirty: true, conflict: draft.base !== textHash(disk), discard: false };
}

const MARK = "● ";

export function dirtyName(name, dirty) {
  const plain = String(name ?? "").startsWith(MARK) ? name.slice(MARK.length) : String(name ?? "");
  return dirty ? MARK + plain : plain;
}

export const baseName = path => String(path ?? "").split(/[\\/]/).pop();

export function parentDir(path) {
  const m = /^(.*)[\\/][^\\/]*$/.exec(String(path ?? ""));
  return m ? m[1] : "";
}

/** The id of the tab with this exact name. */
export function findTab(spaces, name) {
  if (!name) return null;
  for (const s of spaces) for (const t of s.tabs) if (t.name === name) return t.id;
  return null;
}

export function tabInfo(spaces, id) {
  for (const s of spaces) for (const t of s.tabs) if (t.id === id) return t;
  return null;
}

/** The entries of map whose tab still exists. */
export function keepLive(map, spaces, tabOf) {
  const out = {};
  for (const [k, v] of Object.entries(map ?? {})) if (tabInfo(spaces, tabOf(v))) out[k] = v;
  return out;
}
```

```js
// A tab showing one file: text and code are edited and saved, markdown has a rendered view, other files
// offer an external program. With no file yet, it starts as a file search under its folder.
// Unsaved edits are kept as a draft in the app's data folder, because a tab can close without asking.
import { View, div, svg } from "gpui-kit";
import { v_flex, h_flex, Editor, EditorState, TextView, InputState, Input } from "gpui-base";
import { props, spaces, exec, readTextFile, writeTextFile, watchFolder, theme, settings, window, openOverlay, run, notify, shared, setShared, appHome } from "herdr";
import { mkdir, writeFile, unlink } from "fs/promises";
import { editability, previewText, sameText, isSaveKey, restoreEol, syncDecision, saveFailure, REASONS, absolutePath, searchPaths, splitPath, segments, indexCommand, indexLines } from "./files-logic.js";
import { viewKind, textHash, draftName, parseDraft, restoreState, dirtyName, baseName, parentDir, findTab, tabInfo, keepLive } from "./logic.js";
import { OPEN_KEY, OWN_KEY, openKey, openCommands } from "./viewer-open.js";

const LANG = { ts: "js", tsx: "js", mts: "js", cts: "js", jsx: "js", mjs: "js", toml: "yaml", jsonc: "json", svelte: "html", vue: "html" };
const DRAFT_MS = 1000, CLAIM_MS = 500, RESULT_LIMIT = 200;
const GONE = "이 파일을 찾을 수 없어요";
const draftsDir = () => `${appHome()}/data/my-viewer/drafts`;
const draftFile = path => `${draftsDir()}/${draftName(path)}.json`;

export default class Viewer extends View {
  init(_props, cx) {
    const p = props() ?? {};
    this.token = p.token ?? null; this.tabName = p.tabName ?? null; this.on = p.on ?? null; this.root = p.root ?? null;
    const own = this.token ? shared(OWN_KEY)?.[this.token] : null;
    this.path = p.path ?? own?.path ?? null; this.tab = own?.tab ?? null;
    this.buf = null; this.error = null; this.binary = false; this.draftSeq = 0; this.closed = false; this.watching = null;
    this.query = ""; this.at = 0; this.index = null; this.found = null;
    this.search = InputState.new({ placeholder: "이 폴더에서 파일 찾기" });
    this.search.on("change", (_e, cx) => { this.query = this.search.value(); this.at = 0; this.ensureIndex(cx); cx.notify(); });
    this.search.on("submit", (_e, cx) => { this.openResult(cx); cx.notify(); });
    this.claimTab();
    cx.timer.every(CLAIM_MS, () => { if (!this.tab) this.claimTab(); });
    if (this.path) this.load(cx); else this.search.focus();
  }

  // run() returns nothing and props() has no tab id, so the tab is found by the unique name it was opened with.
  claimTab() {
    if (this.tab || !this.token) return;
    const id = findTab(spaces(), this.tabName);
    if (!id) return;
    this.tab = id;
    this.register();
    this.syncName(this.path ? baseName(this.path) : "파일 뷰어");
  }

  register() {
    if (!this.tab) return;
    const list = spaces(), own = keepLive(shared(OWN_KEY), list, v => v.tab);
    setShared(OWN_KEY, { ...own, [this.token]: { tab: this.tab, path: this.path } });
    if (this.path) setShared(OPEN_KEY, { ...keepLive(shared(OPEN_KEY), list, v => v), [openKey(this.on, this.path)]: this.tab });
  }

  syncName(name) {
    if (!this.tab) return;
    const current = tabInfo(spaces(), this.tab)?.name ?? "";
    const want = dirtyName(name ?? current, Boolean(this.buf?.dirty));
    if (want === current) return;
    try { run("tab.rename", { id: this.tab, name: want }); } catch {}
  }

  async readDraft() {
    if (this.on) return null;
    return parseDraft(await readTextFile(draftFile(this.path)).catch(() => null));
  }

  async dropDraft(path = this.path) {
    try { await unlink(draftFile(path)); } catch {} // usually there is none
  }

  async flushDraft() {
    const buf = this.buf, path = this.path;
    if (!buf?.editable) return;
    if (!buf.dirty) return this.dropDraft(path);
    try {
      await mkdir(draftsDir(), { recursive: true });
      await writeFile(draftFile(path), JSON.stringify({ path, text: buf.state.value(), base: textHash(buf.base ?? "") }));
    } catch (e) {
      if (!this.draftWarned) notify({ title: "초안을 보관하지 못했어요", body: String(e), level: "warning", timeout: 5000 });
      this.draftWarned = true;
    }
  }

  scheduleDraft(cx) {
    const seq = ++this.draftSeq;
    cx.timer.after(DRAFT_MS, () => { if (seq === this.draftSeq) this.flushDraft(); });
  }

  makeBuffer(path, text, base) {
    const t = theme(), ext = path.includes(".") ? path.split(".").pop().toLowerCase() : "txt";
    const { editable, reason } = editability({ remote: Boolean(this.on), text });
    const state = EditorState.new({ language: LANG[ext] ?? ext, readonly: !editable, value: previewText(text),
      font_family: settings().font_family, font_size: 13,
      syntax: { keyword: t.ansi5, string: t.ansi2, comment: t.dim, function: t.ansi4, type: t.ansi3, number: t.ansi1, constant: t.ansi6 } });
    const buf = { state, base, dirty: false, conflict: null, editable, reason, saving: false, view: viewKind(path, text) === "markdown" ? "preview" : "edit", rev: 0 };
    if (editable) state.on("change", (_e, cx) => {
      buf.rev++;
      const dirty = !sameText(state.value(), buf.base);
      if (dirty !== buf.dirty) { buf.dirty = dirty; this.syncName(); cx.notify(); }
      this.scheduleDraft(cx);
    });
    return buf;
  }

  load(cx) {
    const path = this.path;
    this.error = null; this.binary = false;
    if (viewKind(path, null) === "binary") { this.binary = true; return; }
    cx.spawn(async cx => {
      try {
        const disk = await readTextFile(path, { on: this.on });
        if (this.path !== path) return;
        if (viewKind(path, disk) === "binary") { this.binary = true; return; }
        const draft = await this.readDraft();
        if (disk === null && !draft) { this.error = GONE; return; }
        const start = restoreState(draft, path, disk);
        if (start.discard) this.dropDraft(path);
        this.buf?.state.release();
        const buf = this.makeBuffer(path, start.text, disk);
        buf.dirty = start.dirty;
        buf.conflict = start.conflict ? "draft" : null;
        this.buf = buf;
        this.syncName();
        this.watch(cx);
      } catch (e) { if (this.path === path) this.error = `파일을 읽지 못했어요: ${e?.message ?? e}`; }
      finally { cx.notify(); }
    });
  }

  watch(cx) {
    if (this.on || this.watching === this.path) return;
    const path = this.path, dir = parentDir(path);
    this.watching = path;
    cx.spawn(async cx => {
      try {
        while (!this.closed && this.path === path) {
          await watchFolder(dir);
          await this.syncDisk();
          cx.notify();
        }
      } catch { if (this.watching === path) this.watching = null; }
    });
  }

  // After a folder change: a clean buffer follows the disk, a dirty one that diverged becomes a conflict.
  async syncDisk() {
    const buf = this.buf;
    if (!buf || buf.saving || buf.conflict === "draft") return;
    const snap = buf.base;
    const disk = await readTextFile(this.path).catch(() => undefined);
    if (disk === undefined || this.buf !== buf || buf.saving || buf.base !== snap || (disk !== null && disk.includes("\0"))) return;
    switch (syncDecision({ base: buf.base, dirty: buf.dirty, value: buf.state.value() }, disk)) {
      case "none": buf.conflict = null; break;
      case "adopt": buf.base = disk; buf.dirty = false; buf.conflict = null; this.syncName(); this.dropDraft(); break;
      case "reload": this.reload(disk); break;
      case "conflict": buf.conflict = "disk"; break;
      case "drop":
        buf.state.release(); this.buf = null; this.error = GONE;
        break;
    }
  }

  // Takes the disk text into a buffer with no edits to keep (recreated if it changes editability).
  reload(disk) {
    const buf = this.buf, next = editability({ remote: Boolean(this.on), text: disk });
    if (next.editable !== buf.editable || next.reason !== buf.reason) {
      buf.state.release();
      const fresh = this.makeBuffer(this.path, disk, disk);
      fresh.view = buf.view;
      this.buf = fresh;
    } else {
      buf.base = disk; buf.dirty = false; buf.conflict = null; buf.rev++;
      buf.state.set_value(previewText(disk));
    }
    this.syncName();
  }

  save(cx) {
    const buf = this.buf;
    if (!buf?.editable || buf.saving || buf.conflict) return;
    const path = this.path, value = restoreEol(buf.state.value(), buf.base), base = buf.base;
    buf.saving = true;
    cx.spawn(async cx => {
      try {
        await writeTextFile(path, value, base);
        buf.base = value; buf.conflict = null; buf.dirty = !sameText(buf.state.value(), value);
        this.draftSeq++; // a pending draft must not come back after the save
        if (!buf.dirty) await this.dropDraft(path);
        this.syncName();
      } catch (e) {
        // writeTextFile rejects stale snapshots and real failures alike: the disk tells them apart.
        const disk = await readTextFile(path).catch(() => undefined);
        if (disk !== undefined && saveFailure(base, disk) === "conflict") buf.conflict = "disk";
        else notify({ title: "저장하지 못했어요", body: `${path}\n${e}`, level: "error", timeout: 5000 });
      } finally { buf.saving = false; cx.notify(); }
    });
  }

  // Conflict: keep my edits, using what's on disk now as the expected snapshot.
  overwrite(cx) {
    const buf = this.buf;
    if (!buf || buf.saving) return;
    cx.spawn(async cx => {
      try {
        const disk = await readTextFile(this.path);
        if (this.buf !== buf) return;
        buf.base = disk; buf.conflict = null;
        this.save(cx);
      } catch (e) { notify({ title: "저장하지 못했어요", body: `${this.path}\n${e}`, level: "error", timeout: 5000 }); }
      finally { cx.notify(); }
    });
  }

  // Draft conflict: go on with the draft, now against what's on disk.
  keepDraft(cx) {
    const buf = this.buf;
    if (!buf) return;
    cx.spawn(async cx => {
      try {
        const disk = await readTextFile(this.path);
        if (this.buf !== buf) return;
        buf.base = disk; buf.conflict = null;
        this.flushDraft();
      } catch (e) { notify({ title: "파일을 읽지 못했어요", body: `${this.path}\n${e}`, level: "error", timeout: 5000 }); }
      finally { cx.notify(); }
    });
  }

  // Conflict: throw my edits away and take the disk.
  loadDisk(cx) {
    const buf = this.buf;
    if (!buf) return;
    this.draftSeq++;
    cx.spawn(async cx => {
      try {
        const disk = await readTextFile(this.path, { on: this.on });
        if (this.buf !== buf) return;
        await this.dropDraft();
        if (disk === null) { buf.state.release(); this.buf = null; this.error = GONE; }
        else this.reload(disk);
      } catch (e) { notify({ title: "파일을 읽지 못했어요", body: `${this.path}\n${e}`, level: "error", timeout: 5000 }); }
      finally { cx.notify(); }
    });
  }

  openExternal() {
    openOverlay({ mod: "my-files-open", kind: "popup", width: 200, height: 136, props: { path: this.path, dir: false } });
  }

  // A new tab: a nano split needs a terminal pane to split, and this tab has none.
  openInNano() {
    try { run("terminal.open", { argv: ["nano", this.path], folder: parentDir(this.path), place: "tab" }); }
    catch (e) { notify({ title: "nano를 열지 못했어요", body: String(e), level: "error", timeout: 5000 }); }
  }

  ensureIndex(cx) {
    if (!this.query.trim() || !this.root || this.index?.building || this.index?.paths) return;
    const index = { paths: null, truncated: false, building: true, error: null }, windows = !this.on && window()?.platform === "windows";
    this.index = index;
    cx.spawn(async cx => {
      try {
        const c = indexCommand(this.root, windows);
        const r = await exec(c.command, c.args, { on: this.on });
        if (r.code !== 0) throw new Error((r.stderr || "").trim() || `exit ${r.code}`);
        Object.assign(index, indexLines(r.stdout, windows));
      } catch (e) { index.error = String(e?.message ?? e); }
      finally { index.building = false; cx.notify(); }
    });
  }

  results() {
    const q = this.query.trim(), paths = this.index?.paths;
    if (!q || !paths) return null;
    if (this.found?.q !== q || this.found.paths !== paths) this.found = { q, paths, value: searchPaths(paths, q, RESULT_LIMIT) };
    return this.found.value;
  }

  openResult(cx) {
    const r = this.results()?.results[this.at];
    if (r) this.pick(r.path, cx);
  }

  // A file chosen in the search. Already open elsewhere: go there and close this empty tab.
  pick(rel, cx) {
    const path = absolutePath(this.root, rel, { remote: Boolean(this.on), windows: !this.on && window()?.platform === "windows" });
    const commands = openCommands({ path, on: this.on, root: this.root, space: null }, shared(OPEN_KEY), spaces(), this.token);
    if (commands[0][0] !== "tab.new") {
      const me = this.tab;
      for (const [name, args] of commands) run(name, args);
      if (me) run("tab.close", { id: me });
      return;
    }
    this.path = path;
    this.register();
    this.syncName(baseName(path));
    this.load(cx);
  }

  async close() {
    this.closed = true;
    this.draftSeq++;
    await this.flushDraft();
  }

  resultRow(r, i, t, cx) {
    const { dir, name } = splitPath(r.path), start = r.path.length - name.length;
    const hits = r.hits.filter(h => h >= start).map(h => h - start);
    return h_flex().id(`found-${r.path}`).h(28).flex_none().px(8).gap(7).items_center().rounded(7).cursor_pointer()
      .when(i === this.at, el => el.bg(t.selected)).hover(s => s.bg(t.hover))
      .on_click((_e, cx) => { this.at = i; this.pick(r.path, cx); cx.notify(); })
      .child(svg("mods/my-files/file.svg").size(14).flex_none().text_color(t.muted))
      .child(h_flex().min_w(0).flex_shrink(1).overflow_hidden()
        .children(segments(name, hits).map(s => div().when(s.hit, el => el.text_color(t.accent).font_semibold()).child(s.text.replaceAll(" ", " ")))))
      .child(div().flex_1().min_w(0).truncate().text_size(11).text_color(t.dim).child(dir));
  }

  pickerView(t, cx) {
    const index = this.index, found = this.results(), q = this.query.trim();
    const note = text => div().p(8).text_size(12).text_color(t.dim).child(text);
    if (found && this.at >= found.results.length) this.at = Math.max(0, found.results.length - 1);
    return v_flex().flex_1().min_h(0)
      .child(h_flex().h(40).flex_none().px(14).gap(8).items_center().border_b(1).border_color(t.border)
        .on_key_down((e, cx) => {
          const n = this.results()?.results.length ?? 0;
          if (e.key === "down" && n) this.at = Math.min(this.at + 1, n - 1);
          else if (e.key === "up" && n) this.at = Math.max(this.at - 1, 0);
          else return;
          cx.stop_propagation();
          cx.notify();
        })
        .child(svg("mods/my-files/search.svg").size(14).flex_none().text_color(t.dim))
        .child(Input.new(this.search).flex_1().h_full().text_size(13).text_color(t.foreground))
        .when(Boolean(index?.building), el => el.child(div().flex_none().text_size(11).text_color(t.dim).child("목록 만드는 중…"))))
      .child(v_flex().id("viewer-found").flex_1().min_h(0).overflow_y_scrollbar().p(6)
        .when(!this.root, el => el.child(note("폴더가 없어요")))
        .when(Boolean(this.root) && !q, el => el.child(note(`${baseName(this.root)} 아래에서 파일 이름을 찾아요`)))
        .when(Boolean(index?.error), el => el.child(div().p(8).text_size(12).text_color(t.ansi1).child(`파일 목록을 만들지 못했어요: ${index.error}`)))
        .when(Boolean(index?.truncated), el => el.child(note("파일이 너무 많아 처음 50,000개에서만 찾아요")))
        .children(found ? found.results.map((r, i) => this.resultRow(r, i, t, cx)) : [])
        .when(Boolean(found?.more), el => el.child(note("더 있어요 · 검색어를 더 입력하세요")))
        .when(Boolean(found) && found.results.length === 0, el => el.child(note(`‘${q}’와 맞는 파일이 없어요`))));
  }

  render(cx) {
    const t = theme(), buf = this.buf, path = this.path;
    const button = (id, label, onClick) => h_flex().id(id).h(24).px(8).flex_none().items_center().rounded(6).cursor_pointer()
      .text_size(12).text_color(t.foreground).border_1().border_color(t.border).hover(s => s.bg(t.hover))
      .on_click((_e, cx) => { onClick(cx); cx.notify(); }).child(label);
    const chip = (id, label, on, onClick) => h_flex().id(id).h(24).px(9).flex_none().items_center().rounded(6).cursor_pointer().text_size(12)
      .bg(on ? t.selected : t.background).text_color(on ? t.foreground : t.muted)
      .hover(s => s.bg(t.hover)).on_click((_e, cx) => { onClick(cx); cx.notify(); }).child(label);
    const bar = () => h_flex().h(34).flex_none().px(14).gap(8).items_center().border_b(1).border_color(t.border).text_size(12);
    const root = v_flex().id("viewer").size_full().min_h(0).overflow_hidden().bg(t.background).text_color(t.foreground).text_size(13);
    if (!path) return root.child(this.pickerView(t, cx));
    const message = text => v_flex().flex_1().items_center().justify_center().gap(10).text_color(t.dim).child(div().child(text));
    return root
      .child(h_flex().h(36).flex_none().px(14).gap(8).items_center().border_b(1).border_color(t.border)
        .child(div().flex_1().min_w(0).truncate().text_size(12).text_color(t.dim).child(`${this.on ? `${this.on}:` : ""}${path}`))
        .when(Boolean(buf?.dirty), el => el.child(div().flex_none().text_size(11).text_color(t.accent).child("● 저장 안 됨")))
        .when(Boolean(buf) && viewKind(path, null) === "markdown", el => el
          .child(chip("viewer-preview", "보기", buf.view === "preview", () => { buf.view = "preview"; }))
          .child(chip("viewer-edit", "편집", buf.view === "edit", () => { buf.view = "edit"; }))))
      .when(Boolean(buf && !buf.editable), el => el.child(bar().text_color(t.dim)
        .child(div().flex_1().min_w(0).truncate().child(REASONS[buf.reason] ?? "읽기 전용"))
        .child(button("viewer-nano", "nano로 열기", () => this.openInNano()))))
      .when(buf?.conflict === "disk", el => el.child(bar()
        .child(div().flex_1().min_w(0).truncate().text_color(t.blocked).child("디스크의 파일이 바뀌었어요"))
        .child(button("viewer-overwrite", "내 것으로 덮어쓰기", cx => this.overwrite(cx)))
        .child(button("viewer-load", "디스크 버전 불러오기", cx => this.loadDisk(cx)))))
      .when(buf?.conflict === "draft", el => el.child(bar()
        .child(div().flex_1().min_w(0).truncate().text_color(t.blocked).child("디스크의 파일이 바뀌었어요 · 저장 안 한 초안이 있어요"))
        .child(button("viewer-keep", "초안 유지", cx => this.keepDraft(cx)))
        .child(button("viewer-load-draft", "디스크 버전 불러오기", cx => this.loadDisk(cx)))))
      .child(this.binary
        ? message(`${baseName(path)} · 여기서 표시할 수 없는 파일이에요`)
          .when(!this.on, el => el.child(button("viewer-external", "외부 프로그램으로 열기", () => this.openExternal())))
        : this.error
          ? message(this.error).child(button("viewer-retry", "다시 읽기", cx => this.load(cx)))
            .when(!this.on, el => el.child(button("viewer-external", "외부 프로그램으로 열기", () => this.openExternal())))
            .when(Boolean(this.on), el => el.child(button("viewer-nano-error", "nano로 열기", () => this.openInNano())))
          : !buf
            ? message("불러오는 중…")
            : buf.view === "preview"
              ? v_flex().id("viewer-md").flex_1().min_h(0).overflow_y_scrollbar().p(18)
                .child(buf.state.value().trim()
                  ? TextView.markdown(`viewer-md-${buf.rev}`, buf.state.value()).selectable(true)
                  : div().text_color(t.dim).child("빈 문서"))
              : v_flex().id("viewer-editor").flex_1().min_h(0)
                .on_key_down((e, cx) => { if (isSaveKey(e.keystroke)) { cx.stop_propagation(); this.save(cx); cx.notify(); } })
                .child(Editor.new(buf.state).flex_1().min_h(0)));
  }
}
```

- [ ] **Step 3: 로드 확인과 실제 탭 시험**

임시 폴더에 시험 파일 하나를 만들고(사용자 파일을 열지 않는다) 설치한 뒤 탭을 연다. Bash에서:

```bash
T="$(cygpath -w "$TEMP")\\viewer-test.txt"; printf 'hello\n' > "$TEMP/viewer-test.txt"
mkdir -p ~/.config/herdr-app/mods/my-viewer && cp mods/my-viewer/* ~/.config/herdr-app/mods/my-viewer/
herdr-app check ~/.config/herdr-app/mods/my-viewer
node -e 'const a={name:"파일 뷰어 viewertest",mod:"my-viewer",props:{path:process.argv[1],on:null,root:null,token:"viewertest",tabName:"파일 뷰어 viewertest"}};require("child_process").execFileSync("herdr-app",["do","tab.new",JSON.stringify(a)],{stdio:"inherit"})' "$T"
sleep 3; herdr-app shot "$TEMP/viewer.png"; herdr-app status
```

Expected: `check passed`. 스크린샷(Read)에서 탭 이름이 `viewer-test.txt`이고 내용 `hello`가 보인다. `herdr-app status`의 errors에 `my-viewer`가 없다. (원격 머신 연결 오류 2건과 `zz-spike` 재로드 오류 1건은 이미 있던 것이다.)

스크린샷의 탭 바와 뷰어 외의 내용은 설명하지 않는다. 끝나면 `herdr-app ls`로 탭 ID를 찾아 `herdr-app do tab.close --id <ID> --confirmed true`로 닫고, 스크린샷과 시험 파일을 지운다.

- [ ] **Step 4: `mods/my-files-drag/main.js`를 다음으로 교체**

```js
// Drag layer for my-files: a transparent full-window overlay opened when a row is dragged out of
// the Files panel. The name follows the pointer. Releasing back over the panel cancels; over the tab
// bar, a file opens in a viewer tab; anywhere else, props.text is typed into the focused terminal
// (clipboard + terminal.paste).
import { View, div } from "gpui-kit";
import { h_flex } from "gpui-base";
import { props, closeOverlay, run, theme, notify, shared, spaces } from "herdr";
import { BOUNDS_KEY, OPEN_KEY, dropAction, openCommands, newToken } from "./viewer-open.js";

export default class MyFilesDrag extends View {
  init() { this.pos = null; this.done = false; }

  action(position) {
    const { panel, dir } = props() ?? {};
    return dropAction({ position, panel, tabs: shared(BOUNDS_KEY), dir });
  }

  finish(position, cx) {
    if (this.done) return;
    this.done = true; cx.notify(); // hides the label at once
    const { text, path, on, overlay } = props() ?? {};
    try {
      const action = this.action(position);
      // ponytail: lands in the focused terminal, not the one under the pointer (panes() has no bounds),
      // and replaces the clipboard.
      if (action === "paste" && text) {
        cx.write_to_clipboard(text);
        run("terminal.paste", {});
      } else if (action === "open" && path) {
        const list = spaces(), space = list.find(s => s.active)?.id;
        for (const [name, args] of openCommands({ path, on: on ?? null, root: null, space }, shared(OPEN_KEY), list, newToken(Date.now(), Math.random()))) run(name, args);
      } else if (action === "folder") {
        notify({ title: "폴더는 열 수 없어요", level: "info", timeout: 3000 });
      }
    } catch (e) {
      notify({ title: "놓은 곳에 넣지 못했어요", body: String(e), level: "error", timeout: 3000 });
    } finally {
      if (overlay) closeOverlay(overlay);
    }
  }

  hint() {
    const action = this.action(this.pos);
    return action === "open" ? "· 놓으면 새 탭으로 열기" : action === "folder" ? "· 폴더는 열 수 없어요" : action === "cancel" ? "· 놓으면 취소" : "· 놓으면 입력";
  }

  render() {
    const t = theme(), { name = "", dir = false, icon } = props() ?? {};
    return div().id("my-files-drag").size_full().bg("#00000001")
      .on_mouse_move((e, cx) => { if (!this.done) { this.pos = e.position; cx.notify(); } })
      .on_mouse_up("left", (e, cx) => this.finish(e.position, cx))
      // Released before this layer opened: the next press just closes it.
      .on_mouse_down("left", (_e, cx) => { cx.stop_propagation(); this.finish(null, cx); })
      .when(Boolean(this.pos) && !this.done, el => el.child(
        h_flex().absolute().left(this.pos.x + 14).top(this.pos.y + 12).h(26).px(10).gap(6).items_center()
          .rounded(7).bg(t.popover).border_1().border_color(t.accent).text_color(t.foreground).text_size(12)
          .child(icon ?? (dir ? "📁" : "📄")).child(name)
          .child(div().text_color(t.dim).child(this.hint()))));
  }
}
```
